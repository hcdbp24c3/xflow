class Sent {
  final String method;
  final String url;
  final String? body;
  final Map<String, String> headers;
  Sent(this.method, this.url, this.body, this.headers);
}

final sent = <Sent>[];
http.Response Function(http.Request request) responder =
    (_) => http.Response('not found', 404);

http.Response utf8Json(String body, [int status = 200]) => http.Response.bytes(
    utf8.encode(body), status,
    headers: {'content-type': 'application/json; charset=utf-8'});

final fakeClient = MockClient((request) async {
  if (request.url.host != 'x.com') {
    // the transaction id generator: never answered
    return http.Response('no', 404);
  }
  sent.add(Sent(request.method, request.url.toString(),
      request.method == 'POST' ? request.body : null, request.headers));
  return responder(request);
});

Future<T> run<T>(Future<T> Function() body) =>
    http.runWithClient(body, () => fakeClient);

/// The address split so that the long constant parts are stored once:
/// path + [name, encoded value] pairs, exactly as they stand in the URL.
final Map<String, String> blobs = {}; // encoded value -> blob name
Map<String, dynamic> splitUrl(String url) {
  final q = url.indexOf('?');
  final base = q < 0 ? url : url.substring(0, q);
  final params = <List<dynamic>>[];
  if (q >= 0) {
    for (final pair in url.substring(q + 1).split('&')) {
      final eq = pair.indexOf('=');
      final name = pair.substring(0, eq);
      final value = pair.substring(eq + 1);
      if (name == 'variables') {
        params.add([name, value]);
      } else {
        final blob = blobs.putIfAbsent(value, () => 'blob${blobs.length + 1}');
        params.add([
          name,
          {'blob': blob}
        ]);
      }
    }
  }
  // must rebuild to the very same text
  final rebuilt = StringBuffer(base);
  for (var i = 0; i < params.length; i++) {
    rebuilt.write(i == 0 ? '?' : '&');
    final v = params[i][1];
    rebuilt.write('${params[i][0]}=');
    rebuilt.write(v is String
        ? v
        : blobs.entries.firstWhere((e) => e.value == v['blob']).key);
  }
  if (rebuilt.toString() != url) throw StateError('split lost bytes: $url');
  return {'base': base, 'params': params};
}

List<Map<String, dynamic>> takeSent() {
  final list = sent
      .map((s) => {
            'method': s.method,
            ...splitUrl(s.url),
            if (s.body != null) 'body': s.body,
          })
      .toList();
  sent.clear();
  return list;
}

Map<String, dynamic> tweetJson(Tweet t) => {
      'id': t.id,
      'text': t.text,
      'userHandle': t.userHandle,
      'userAvatarUrl': t.userAvatarUrl,
      'mediaKey': t.mediaKey,
      'mediaUrls': t.mediaUrls,
      'thumbnailUrl': t.thumbnailUrl,
      'isVideo': t.isVideo,
      'createdAtMillis': t.createdAt?.millisecondsSinceEpoch,
      'createdAtIsUtc': t.createdAt?.isUtc,
      'source': t.source,
      'isLiked': t.isLiked,
      'favoriteCount': t.favoriteCount,
      'replyCount': t.replyCount,
      'retweetCount': t.retweetCount,
      'isRetweeted': t.isRetweeted,
      'viewCount': t.viewCount,
      'isBookmarked': t.isBookmarked,
    };

Map<String, dynamic> responseJson(TweetResponse r) => {
      'tweets': r.tweets.map(tweetJson).toList(),
      'cursorTop': r.cursorTop,
      'cursorBottom': r.cursorBottom,
    };

Map<String, dynamic>? subJson(Subscription? s) => s == null
    ? null
    : {
        'id': s.id,
        'screenName': s.screenName,
        'name': s.name,
        'profileImageUrl': s.profileImageUrl,
        'description': s.description,
        'followersCount': s.followersCount,
        'followingCount': s.followingCount,
      };

String? filterNames(Set<MediaFilter>? f) => f?.map((e) => e.name).join(',');

void writeJson(String name, Object? value) {
  File('$outDir/$name').writeAsStringSync(
      '${const JsonEncoder.withIndent(' ').convert(value)}\n');
}
